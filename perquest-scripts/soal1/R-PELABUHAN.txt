! Soal 1 - Konfigurasi IP R-PELABUHAN (K2, skema CIDR)
! Modul: HWIC-2T di slot 3 (Serial0/3/x). S0/3/0 = jalur ke router atas (sisi DTE).
enable
configure terminal
hostname R-PELABUHAN
no ip domain-lookup
interface g0/0
 description LAN Sensor dan SCADA Pelabuhan
 ip address 10.18.64.1 255.255.224.0
 no shutdown
interface g0/1
 description LAN Kantor Pelabuhan
 ip address 10.18.96.1 255.255.255.128
 no shutdown
interface s0/3/0
 description L6 ke R-UTARA
 ip address 10.18.96.130 255.255.255.252
 no shutdown
end
copy running-config startup-config
