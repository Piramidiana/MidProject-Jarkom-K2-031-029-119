! Soal 1 - Konfigurasi IP R-SELATAN (K2, skema CIDR)
! Modul: HWIC-2T di slot 3 (Serial0/3/x). S0/3/0 = jalur ke router atas (sisi DTE).
enable
configure terminal
hostname R-SELATAN
no ip domain-lookup
interface g0/0
 description LAN WiFi Publik Selatan
 ip address 10.16.192.1 255.255.224.0
 no shutdown
interface g0/1
 description LAN PJU Pintar Selatan
 ip address 10.16.224.1 255.255.255.128
 no shutdown
interface s0/3/0
 description L3 ke R-PUSAT
 ip address 10.16.224.130 255.255.255.252
 no shutdown
end
copy running-config startup-config
